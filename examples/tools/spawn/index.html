  // The model likes to wrap output in a fence no matter how firmly it is asked not to.
  function stripFence(s) {
    const m = s.match(/```(?:html)?\s*\n([\s\S]*?)```/);
    return (m ? m[1] : s).trim();
  }

  $("go").addEventListener("click", async () => {
    const name = $("name").value.trim().toLowerCase();
    const idea = $("idea").value.trim();
    if (!name) return setStatus("Give it a name first.", "err");
    if (!/^[a-z0-9][a-z0-9-]{0,62}$/.test(name)) return setStatus("Names are lowercase letters, digits and dashes.", "err");
    if (!idea) return setStatus("Say what it should do.", "err");
    if (!chosen.size) return setStatus("Pick at least one primitive.", "err");

    $("go").disabled = true;
    $("deploy").disabled = true;
    $("download").disabled = true;
    $("live").hidden = true;
    $("outCard").hidden = false;
    $("out").textContent = "";
    setStatus("Writing…");

    try {
      const res = await worlds.ai.complete({
        prompt: buildPrompt(name, idea),
        model: "smart",
        max_tokens: 8000,
        stream: true,
        onToken: (t) => { $("out").textContent += t; $("out").scrollTop = $("out").scrollHeight; },
      });
      generated = stripFence(res.text || $("out").textContent);
      $("out").textContent = generated;
      if (!/<script[^>]*src=["']\/worlds\.js/.test(generated)) {
        setStatus("Written, but it left out the /worlds.js tag — check before deploying.", "err");
      } else {
        setStatus(`Written — ${generated.length.toLocaleString()} characters.`, "ok");
      }
      $("deploy").disabled = false;
      $("download").disabled = false;
    } catch (e) {
      setStatus(`${e.code || "error"}: ${e.message}`, "err");
    } finally {
      $("go").disabled = false;
    }
  });

  $("deploy").addEventListener("click", async () => {
    if (!generated) return;
    const name = $("name").value.trim().toLowerCase();
    $("deploy").disabled = true;
    setStatus("Deploying…");
    try {
      // Same endpoint the homepage drag-drop uses: each form field name is the file's
      // path, and the value is its bytes.
      const form = new FormData();
      form.set("site", name);
      form.set("index.html", new Blob([generated], { type: "text/html" }), "index.html");
      form.set(".world.json", new Blob([JSON.stringify({
        description: $("idea").value.trim().slice(0, 140),
        category: "experiments",
      })], { type: "application/json" }), ".world.json");

      const res = await fetch("/api/v1/deploy-folder", {
        method: "POST",
        headers: { "x-worlds-csrf": "1" },
        body: form,
      });
      const data = await res.json().catch(() => ({}));
      if (!res.ok) {
        const err = data.error || {};
        throw Object.assign(new Error(err.message || res.statusText), { code: err.code });
      }
      setStatus("Live.", "ok");
      const url = data.url || `/app/${name}/`;
      $("live").hidden = false;
      $("live").innerHTML = `<a href="${esc(url)}" target="_blank" rel="noopener">Open ${esc(name)} →</a>`;
      toast("Deployed");
    } catch (e) {
      setStatus(`${e.code || "error"}: ${e.message}`, "err");
      $("deploy").disabled = false;
    }
  });

  $("download").addEventListener("click", () => {
    if (!generated) return;
    const a = document.createElement("a");
    a.href = URL.createObjectURL(new Blob([generated], { type: "text/html" }));
    a.download = "index.html";
    a.click();
    URL.revokeObjectURL(a.href);
  });

  renderPrims();
  worlds.me().then((me) => { $("whoami").textContent = me.name || me.handle; }).catch(() => {});
})();
